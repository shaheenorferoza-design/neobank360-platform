
CREATE OR REPLACE TABLE dim_customer (
    -- Surrogate Key
    customer_key            NUMBER GENERATED ALWAYS AS IDENTITY,
    -- Business Key
    customer_id             VARCHAR(50) NOT NULL,
    -- Customer Attributes
    full_name               VARCHAR(200),
    date_of_birth           DATE,
    age_band                VARCHAR(50),
    city                    VARCHAR(100),
    state                   VARCHAR(100),
    occupation              VARCHAR(100),
    -- Compliance / Risk Attributes
    kyc_status              VARCHAR(50),
    risk_category           VARCHAR(50),
    -- Relationship Metadata
    customer_since_date     DATE,
    -- SCD Type 2 Columns
    effective_start_date    TIMESTAMP,
    effective_end_date      TIMESTAMP,
    current_flag            VARCHAR(1),
    -- Warehouse Audit Columns
    warehouse_insert_ts     TIMESTAMP DEFAULT CURRENT_TIMESTAMP(),
    CONSTRAINT pk_dim_customer
        PRIMARY KEY (customer_key)
);
